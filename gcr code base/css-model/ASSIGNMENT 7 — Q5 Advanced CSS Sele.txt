CSS Code
/* Attribute selectors */
[type="text"] { border: 2px solid #2563eb; padding: 10px; }
/* Attribute prefix selector */
[href^="https"] { color: #15803d; font-weight: bold; }
/* Pseudo-classes */
.items li:hover { background: #dbeafe; }
.items li:nth-child(2) { font-weight: bold; }
input:focus { outline: 3px solid #f59e0b; }
/* Pseudo-elements */
.intro::before { content: "H "; }
.intro::after { content: " H"; }
/* Child combinator > */
.parent > span { color: #dc2626; }
/* Adjacent sibling + */
.parent span + p { margin-top: 20px; }
/* General sibling ~ */
.parent span ~ em { color: #7c3aed; }
/* Descendant combinator */
.parent p b { text-decoration: underline; }
body { font-family: Arial, sans-serif; padding: 35px; line-height: 1.7; }
